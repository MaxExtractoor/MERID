    <script>
        let sessionsChart, memoryChart, anomaliesHeatmap, cpuChart, storageChart;

        function showTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(content => content.style.display = 'none');
            document.getElementById(tabId + '-tab').style.display = 'block';
            document.querySelectorAll('.tab-button').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
        }

        function initCharts() {
            sessionsChart = new Chart(document.getElementById('sessions-pie'), {
                type: 'pie',
                data: {
                    labels: ['Active', 'Idle'],
                    datasets: [{
                        data: [60, 40],
                        backgroundColor: ['#5cb85c', '#d3d3d3']
                    }]
                },
                options: { responsive: true }
            });

            memoryChart = new Chart(document.getElementById('memory-line'), {
                type: 'line',
                data: {
                    labels: Array.from({length: 10}, (_, i) => `T${i}`),
                    datasets: [{
                        label: 'Memory %',
                        data: Array(10).fill(0),
                        borderColor: '#337ab7'
                    }]
                },
                options: { responsive: true, scales: { y: { beginAtZero: true } } }
            });

            anomaliesHeatmap = new Chart(document.getElementById('anomalies-heatmap'), {
                type: 'matrix',
                data: {
                    datasets: [{
                        label: 'Data Density',
                        data: generateHeatmapData(),
                        backgroundColor(c) { return `hsl(${240 - c.raw * 240}, 75%, 60%)`; },
                        width: 30,
                        height: 30
                    }]
                },
                options: { responsive: true, scales: { x: { type: 'linear' }, y: { type: 'linear' } } }
            });

            cpuChart = new Chart(document.getElementById('cpu-line'), {
                type: 'line',
                data: {
                    labels: Array.from({length: 10}, (_, i) => `T${i}`),
                    datasets: [{
                        label: 'CPU %',
                        data: Array(10).fill(0),
                        borderColor: '#d9534f'
                    }]
                },
                options: { responsive: true, scales: { y: { beginAtZero: true } } }
            });

            storageChart = new Chart(document.getElementById('storage-line'), {
                type: 'line',
                data: {
                    labels: Array.from({length: 10}, (_, i) => `T${i}`),
                    datasets: [{
                        label: 'Storage (MB)',
                        data: Array(10).fill(0),
                        borderColor: '#f0ad4e'
                    }]
                },
                options: { responsive: true, scales: { y: { beginAtZero: true } } }
            });
        }

        function generateHeatmapData() {
            const data = [];
            for (let x = 0; x < 10; x++) {
                for (let y = 0; y < 10; y++) {
                    data.push({ x, y, v: Math.random() });
                }
            }
            return data;
        }

        function updateRealTimeData() {
            // Simulate data
            const activeSessions = Math.floor(Math.random() * 50) + 50;
            const idleSessions = 100 - activeSessions;
            const memory = Math.random() * 100;
            const cpu = Math.random() * 100;
            const storage = Math.random() * 100 + 20;
            const anomaliesData = generateHeatmapData();

            // Update pie
            sessionsChart.data.datasets[0].data = [activeSessions, idleSessions];
            sessionsChart.update();

            // Shift and push line data
            memoryChart.data.datasets[0].data.shift();
            memoryChart.data.datasets[0].data.push(memory);
            memoryChart.update();

            cpuChart.data.datasets[0].data.shift();
            cpuChart.data.datasets[0].data.push(cpu);
            cpuChart.update();

            storageChart.data.datasets[0].data.shift();
            storageChart.data.datasets[0].data.push(storage);
            storageChart.update();

            // Update heatmap
            anomaliesHeatmap.data.datasets[0].data = anomaliesData;
            anomaliesHeatmap.update();

            // Update table
            const tableBody = document.querySelector('#top-agents-table tbody');
            tableBody.innerHTML = '';
            ['Eyes', 'Brain', 'Immune', 'Perception'].forEach(agent => {
                const row = document.createElement('tr');
                row.innerHTML = `
                    <td>${agent}</td>
                    <td>${Math.random() > 0.5 ? 'Active' : 'Idle'}</td>
                    <td>${Math.random().toFixed(2)}</td>
                `;
                tableBody.appendChild(row);
            });
        }

        function runQuery() {
            const query = document.getElementById('query-editor').value;
            document.getElementById('query-result').innerHTML = `Executed: ${query}<br>Result: Simulated data returned.`;
        }

        function simulateProcessing(query) {
            document.getElementById('spinner').style.display = 'block';
            setTimeout(() => {
                document.getElementById('raw').style.display = 'block';
                document.getElementById('raw').textContent = `Analyzing "${query}"... invariants checked... no risks.`;
                document.getElementById('distilled').innerHTML = `<strong>Distilled Output:</strong><br>Query: "${query}"<br>Status: Nominal. Habit Nudge: Review agents regularly.`;
                document.getElementById('spinner').style.display = 'none';
            }, 2000);
        }

        document.getElementById('lockdown').addEventListener('click', () => {
            alert('System Locked Down - All operations halted.');
            document.body.classList.add('amber-shift');
        });

        document.getElementById('user-input').addEventListener('keypress', e => {
            if (e.key === 'Enter' && e.target.value.trim()) {
                simulateProcessing(e.target.value.trim());
                e.target.value = '';
            }
        });

        initCharts();
        setInterval(updateRealTimeData, 3000); // Real-time update
    </script>
